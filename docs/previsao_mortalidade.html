function createChart(data, title) {

  const ptBRLocaleDefinition = {
    decimal: ",",
    thousands: "",
    grouping: [3],
    currency: ["R$", ""]
  };
  d3.formatDefaultLocale(ptBRLocaleDefinition);

  const width = 760;
  const height = 500;
  const margin = {top: 30, right: 20, bottom: 40, left: 50};
  
  const svg = d3.create("svg")
    .attr("viewBox", [0, 0, width, height])
      .attr("width", width)
      .attr("height", height)
      .attr("style", "max-width: 100%; height: auto; height: intrinsic; font: 10px sans-serif;")
      .style("-webkit-tap-highlight-color", "transparent")
      .style("overflow", "visible");

  // Escalas
  const x = d3.scalePoint()
    .domain([...new Set(data.map(d => d.faixa))])
    .range([margin.left, width - margin.right])
    .padding(0.5);

  const y = d3.scaleLinear()
    .domain(d3.extent(data, d => d.prev)).nice()
    .range([height - margin.bottom, margin.top]);

  // Eixos
  svg.append("g")
    .attr("transform", `translate(0,${height - margin.bottom})`)
    .call(d3.axisBottom(x))
    .selectAll("text")
    .style("text-anchor", "end")
    .attr("dx", "-.8em")
    .attr("dy", ".15em")
    .attr("transform", "rotate(-45)");

  svg.append("text")
    .attr("text-anchor", "middle")
    .attr("x", width / 2)
    .attr("y", height) // Posição abaixo do eixo
    .style("font-size", "14px")
    .style("fill", "currentColor")
    .text("Faixa Etária");
    

  svg.append("g")
    .attr("transform", `translate(${margin.left},0)`)
    .call(d3.axisLeft(y).ticks(height / 40))
    .call(g => g.select(".domain").remove())
    .style("font-size", "12px")
    .call(g => g.selectAll(".tick line").clone()
      .attr("x2", width - margin.left - margin.right)
      .attr("stroke-opacity", 0.1))
    .call(g => g.append("text")
      .attr("x", -margin.left)
      .attr("y", 15)
      .attr("fill", "currentColor")
      .attr("text-anchor", "start")
      .style("font-size", "18px")
      .text("log(Mx)"));
  

  // Gradiente e cores
  const colorScale = d3.scaleSequential(d3.interpolateViridis)
    .domain([2024, 2070]);

  const defs = svg.append("defs");
  const nestedData = d3.groups(data, d => d.ano);

  nestedData.forEach(([year, values]) => {
    const gradientId = `gradient-${title}-${year}`;
    
    const gradient = defs.append("linearGradient")
      .attr("id", gradientId)
      .attr("gradientUnits", "userSpaceOnUse")
      .attr("x1", margin.left)
      .attr("y1", 0)
      .attr("x2", width - margin.right)
      .attr("y2", 0);

    gradient.append("stop")
      .attr("offset", "0%")
      .attr("stop-color", colorScale(year));

    gradient.append("stop")
      .attr("offset", "100%")
      .attr("stop-color", colorScale(Math.min(year + 10, 2070)));
  });

  // Linhas
  const line = d3.line()
    .x(d => x(d.faixa))
    .y(d => y(d.prev));

  nestedData.forEach(([year, values]) => {
    svg.append("path")
      .datum(values)
      .attr("fill", "none")
      .attr("stroke", `url(#gradient-${title}-${year})`)
      .attr("stroke-width", 2.5)
      .attr("d", line);
  });

  // Legenda
  const legendWidth = 200;
  const legendHeight = 20;
  const legendX = width - margin.right - legendWidth;
  const legendY = margin.top;

  const legendGradient = defs.append("linearGradient")
    .attr("id", `legend-${title}`)
    .attr("x1", "0%")
    .attr("y1", "0%")
    .attr("x2", "100%")
    .attr("y2", "0%");

  const years = [2024, 2035, 2045, 2055, 2070];
  years.forEach((year, i) => {
    legendGradient.append("stop")
      .attr("offset", `${i * 100 / (years.length - 1)}%`)
      .attr("stop-color", colorScale(year));
  });

  svg.append("rect")
    .attr("x", legendX)
    .attr("y", legendY)
    .attr("width", legendWidth)
    .attr("height", legendHeight)
    .style("fill", `url(#legend-${title})`);

  svg.append("text")
    .attr("x", legendX)
    .attr("y", legendY - 5)
    .attr("font-size", "12px")
    .text("Anos (2024-2070)");

  years.forEach((year, i) => {
    svg.append("text")
      .attr("x", legendX + (i * legendWidth / (years.length - 1)))
      .attr("y", legendY + legendHeight + 15)
      .attr("text-anchor", "middle")
      .attr("font-size", "10px")
      .text(year);
  });

  // Título
  svg.append("text")
    .attr("x", width / 2)
    .attr("y", 20)
    .attr("text-anchor", "middle")
    .style("font-size", "1.2em")
    .text(title);

  return svg.node();
}

function createboxplot(data, metrica) {
  // Configurações
  const margin = ({top: 30, right: 20, bottom: 40, left: 50})
  const width = 760
  const height = 500

  // Calcular domínio do Y baseado nos dados reais
  const yMin = d3.min(plotData, d => d.valor)
  const yMax = d3.max(plotData, d => d.valor)
  const yPadding = (yMax - yMin) * 0.1

  // Escalas
  const x = d3.scaleBand()
    .domain(["ARIMA+ETS", "Lee-Carter", "Lee-Miller"])
    .range([margin.left, width - margin.right])
    .padding(0.3)

  const y = d3.scaleLinear()
    .domain([yMin - yPadding, yMax + yPadding])
    .range([height - margin.bottom, margin.top])

  const color = d3.scaleOrdinal()
    .domain(modelos) // Mapeia os nomes dos modelos...
    .range(['#e41a1c', '#377eb8', '#4daf4a']); // ...para as cores desejadas.

  // Criar SVG
  const svg = d3.create("svg")
    .attr("width", width)
    .attr("height", height)
    .attr("viewBox", [0, 0, width, height])
    .attr("style", "max-width: 100%; height: auto; height: intrinsic; font: 10px sans-serif;")
    .style("-webkit-tap-highlight-color", "transparent")
    .style("background", "#fffdfdff");

  // Função para calcular estatísticas do boxplot
  function boxQuartiles(values) {
    const sorted = values.map(d => d.valor).sort(d3.ascending);
    const q1 = d3.quantile(sorted, 0.25);
    const q2 = d3.quantile(sorted, 0.5);
    const q3 = d3.quantile(sorted, 0.75);
    const iqr = q3 - q1;
    const min = Math.max(d3.min(sorted), q1 - 1.5 * iqr);
    const max = Math.min(d3.max(sorted), q3 + 1.5 * iqr);
    return {q1, q2, q3, min, max};
  }

  // Agrupar dados por modelo
  const grouped = d3.group(plotData, d => d.modelo);

  // Desenhar boxplots
  grouped.forEach((values, modelo) => {
    const stats = boxQuartiles(values);
    const xPos = x(modelo) + x.bandwidth() / 2;
    const boxWidth = x.bandwidth() * 0.6;
    
    const g = svg.append("g");
    
    // Whiskers
    g.append("line")
      .attr("x1", xPos).attr("x2", xPos)
      .attr("y1", y(stats.min)).attr("y2", y(stats.q1))
      .attr("stroke", "black").attr("stroke-width", 1.5); // <== Alterado para preto
      
    g.append("line")
      .attr("x1", xPos).attr("x2", xPos)
      .attr("y1", y(stats.max)).attr("y2", y(stats.q3))
      .attr("stroke", "black").attr("stroke-width", 1.5); // <== Alterado para preto
      
    // Caps dos whiskers
    g.append("line")
      .attr("x1", xPos - boxWidth/4).attr("x2", xPos + boxWidth/4)
      .attr("y1", y(stats.min)).attr("y2", y(stats.min))
      .attr("stroke", "black").attr("stroke-width", 1.5); // <== Alterado para preto
      
    g.append("line")
      .attr("x1", xPos - boxWidth/4).attr("x2", xPos + boxWidth/4)
      .attr("y1", y(stats.max)).attr("y2", y(stats.max))
      .attr("stroke", "black").attr("stroke-width", 1.5); // <== Alterado para preto
      
    // Caixa
    g.append("rect")
      .attr("x", xPos - boxWidth/2)
      .attr("y", y(stats.q3))
      .attr("width", boxWidth)
      .attr("height", y(stats.q1) - y(stats.q3))
      .attr("fill", color(modelo)) // Mantém o preenchimento colorido
      .attr("fill-opacity", 0.7)
      .attr("stroke", "black") // <== Alterado para preto
      .attr("stroke-width", 1.5);
      
    // Mediana
    g.append("line")
      .attr("x1", xPos - boxWidth/2).attr("x2", xPos + boxWidth/2)
      .attr("y1", y(stats.q2)).attr("y2", y(stats.q2))
      .attr("stroke", "black") // <== Alterado para preto
      .attr("stroke-width", 2);
      
    // Pontos individuais (jitter)
    values.forEach(d => {
      svg.append("circle")
        .attr("cx", xPos + (Math.random() - 0.5) * boxWidth * 0.8)
        .attr("cy", y(d.valor))
        .attr("r", 2.5)
        .attr("fill", "gray") // Mantém o preenchimento colorido
        .attr("fill-opacity", 0.5)
        .attr("stroke", "black") // <== Alterado para preto
        .attr("stroke-width", 0.5); // Espessura menor para o contorno dos pontos
    });
  });

  // Eixos
  svg.append("g")
    .attr("transform", `translate(0,${height - margin.bottom})`)
    .call(d3.axisBottom(x));

  svg.append("g")
    .attr("transform", `translate(${margin.left},0)`)
    .call(d3.axisLeft(y));

  // Título do eixo Y
  svg.append("text")
    .attr("transform", "rotate(-90)")
    .attr("y", 0 - margin.left)
    .attr("x", 0 - (height / 2))
    .attr("dy", "1em")
    .style("text-anchor", "middle")
    .text(metrica);

  // Retornar o SVG
  return svg.node();
  
}

function grafico_barra(data){
  const margin = ({top: 30, right: 20, bottom: 40, left: 50})
  const width = 500
  const height = 800

  

  // Lista dos subgrupos (modelos)
  const subgroups = [...new Set(data.map(d => d.modelo))].sort()

  // Lista dos grupos (faixas etárias) - mantendo a ordem original
  const groups = [...new Set(data.map(d => d.faixa))]

  // Criar o SVG
  const svg = d3.create("svg")
    .attr("width", width + margin.left + margin.right)
    .attr("height", height + margin.top + margin.bottom)
    .attr("style", "height: auto; height: intrinsic; font: 10px sans-serif;")
    .style("-webkit-tap-highlight-color", "transparent")

  const g = svg.append("g")
    .attr("transform", `translate(${margin.left},${margin.top})`)

  // Escalas
  // Escala Y para as faixas etárias (grupos)
  const y = d3.scaleBand()
    .domain(groups)
    .range([0, height])
    .padding(0.2)

  // Escala X para os valores
  const x = d3.scaleLinear()
    .domain([d3.min(data, d => d.valor)-0.5, d3.max(data, d => d.valor)])
    .range([0, width])

  // Escala Y para subgrupos (posição das barras dentro de cada faixa etária)
  const ySubgroup = d3.scaleBand()
    .domain(subgroups)
    .range([0, y.bandwidth()])
    .padding(0.05)

  // Escala de cores
  const color = d3.scaleOrdinal()
    .domain(subgroups)
    .range(['#e41a1c', '#377eb8', '#4daf4a'])

  // Adicionar eixo X
  g.append("g")
    .attr("transform", `translate(0,${height})`)
    .call(d3.axisBottom(x))

  // Adicionar eixo Y
  g.append("g")
    .call(d3.axisLeft(y))

  // Criar as barras
  g.append("g")
    .selectAll("g")
    .data(groups)
    .enter()
    .append("g")
    .attr("transform", d => `translate(0,${y(d)})`)
    .selectAll("rect")
    .data(faixa => {
      return subgroups.map(modelo => {
        const dataPoint = data.find(d => d.faixa === faixa && d.modelo === modelo);
        return {
          modelo: modelo,
          valor: dataPoint ? dataPoint.valor : 0,
          faixa: faixa
        };
      });
    })
    .enter()
    .append("rect")
    .attr("x", 0)
    .attr("y", d => ySubgroup(d.modelo))
    .attr("width", d => x(d.valor))
    .attr("height", ySubgroup.bandwidth())
    .attr("fill", d => color(d.modelo))

  // Adicionar rótulos dos eixos
  g.append("text")
    .attr("transform", "rotate(-90)")
    .attr("y", 0 - margin.left)
    .attr("x", 0 - (height / 2))
